직원 로그인 정보 담을 테이블 생성
CREATE TABLE est_company_db.tb_employee (
    emp_id INT AUTO_INCREMENT PRIMARY KEY,
    login_id VARCHAR(50) UNIQUE NOT NULL,      -- 직원 로그인 아이디
    login_pw VARCHAR(255) NOT NULL,            -- 암호화된 비밀번호 (웹 서버에서 Hash 처리)
    department VARCHAR(50) NOT NULL,           -- 부서 (dev, hr, sales, admin)
    role_level VARCHAR(20) DEFAULT 'USER',     -- 권한 (user, manager)
    emp_name VARCHAR(50) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

firewall-cmd --permanent --add-rich-rule='rule family="ipv4" source address="172.30.1.7" port protocol="tcp" port="3306" accept'
firewall-cmd --reload

vi /etc/my.cnf.d/mariadb-server.cnf (수정)
[mysqld]
server-id = 1
log-bin = mariadb-bin

systemctl restart mariadb
mysql -u root -p
CREATE USER 'repl_user'@'172.30.1.7' IDENTIFIED BY 'estwebre!2026';
GRANT REPLICATION SLAVE ON *.* TO 'repl_user'@'172.30.1.7';
FLUSH PRIVILEGES;
SHOW MASTER STATUS;
